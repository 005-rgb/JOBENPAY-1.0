## 2. BACKEND IMPLEMENTATION (MCP SERVER)

**File:** `server/src/routes/mcp.js`

MCP Server harus menangani request `POST` dengan protokol JSON-RPC 2.0.

```javascript
import express from 'express';
import { validateAgentToken } from '../services/agent_auth_service.js';
import { proposeTransaction } from '../services/decision_engine.js';

const router = express.Router();

// Middleware Auth Khusus Agen
async function authenticateAgent(req, res, next) {
  try {
    const token = req.headers.authorization?.replace('Bearer ', '');
    const env = req.headers['x-jobenpay-environment'];

    if (!token || !env) throw new Error('MISSING_CREDENTIALS');

    // Verifikasi Token + Environment
    const agent = await validateAgentToken(token, env);
    req.agent = agent;
    req.environment = env;
    next();
  } catch (err) {
    res.status(401).json({ jsonrpc: '2.0', error: { code: -32000, message: err.message }, id: null });
  }
}

router.post('/mcp', authenticateAgent, async (req, res) => {
  const { jsonrpc, id, method, params } = req.body;

  if (jsonrpc !== '2.0') {
    return res.status(400).json({ jsonrpc: '2.0', error: { code: -32600, message: 'Invalid Request' }, id });
  }

  try {
    if (method === 'initialize') {
      return res.json({ 
        jsonrpc: '2.0', id, 
        result: { protocolVersion: '2026-07-28', capabilities: { tools: {} } } 
      });
    }

    if (method === 'tools/list') {
      // Hanya kembalikan tool yang diizinkan untuk agen ini
      const allowedTools = await getToolsForAgent(req.agent.allowed_tools);
      return res.json({ jsonrpc: '2.0', id, result: { tools: allowedTools } });
    }

    if (method === 'tools/call') {
      const { name, arguments: args } = params;

      // Cek Otorisasi Tool
      if (!req.agent.allowed_tools.includes(name)) {
        return res.status(403).json({ jsonrpc: '2.0', error: { code: -32601, message: 'Tool not allowed' }, id });
      }

      if (name === 'propose_payout') {
        // Panggil Decision Engine (PRD 1.2) dengan Environment yang benar
        const result = await proposeTransaction({
          orgId: req.agent.org_id,
          agentId: req.agent.agent_id,
          environment: req.environment, // WAJIB
          payeeId: args.payeeId,
          amountMinor: args.amountMinor,
          currency: args.currency
        });

        return res.json({ 
          jsonrpc: '2.0', id, 
          result: { status: result.status, decision: result.decision, reasonCodes: result.reason_codes } 
        });
      }
    }

    res.status(404).json({ jsonrpc: '2.0', error: { code: -32601, message: 'Method not found' }, id });

  } catch (err) {
    res.status(500).json({ jsonrpc: '2.0', error: { code: -32603, message: 'Internal error' }, id });
  }
});

export default router;
```

## 3. FRONTEND IMPLEMENTATION (WEBMCP BRIDGE)

**File:** `web/src/lib/webmcp_bridge.js`

Ini berjalan di browser. Agen AI (seperti Gemini di Chrome) akan mendeteksi tool ini melalui `navigator.modelContext`.

```javascript
export function initWebMCP() {
  if (!('modelContext' in navigator)) {
    console.warn('WebMCP not supported in this browser.');
    return;
  }

  // Tool 1: Mencari Vendor
  navigator.modelContext.registerTool({
    name: 'search_vendors',
    description: 'Search for approved payees in JOBENPAY.',
    inputSchema: { type: 'object', properties: { query: { type: 'string' } } },
    execute: async (args) => {
      const env = localStorage.getItem('currentEnvironment') || 'SANDBOX';
      const res = await fetch(`/api/v1/payees?search=${args.query}`, {
        headers: { 'X-Jobenpay-Environment': env }
      });
      return await res.json();
    }
  });

  // Tool 2: Inisiasi Payout (Hanya usulan)
  navigator.modelContext.registerTool({
    name: 'initiate_payout',
    description: 'Propose a payout to a vendor. Requires human approval.',
    inputSchema: { 
      type: 'object', 
      properties: { 
        payeeId: { type: 'string' }, 
        amountMinor: { type: 'string' }, 
        currency: { type: 'string' } 
      } 
    },
    execute: async (args) => {
      const env = localStorage.getItem('currentEnvironment') || 'SANDBOX';
      const res = await fetch('/api/v1/requests', {
        method: 'POST',
        headers: { 
          'Content-Type': 'application/json',
          'X-Jobenpay-Environment': env // WAJIB: Menyuntikkan environment aktif
        },
        body: JSON.stringify(args)
      });
      return await res.json();
    }
  });
}
```

### 💡 Mengapa Prompt Ini Sangat Kuat & Detail?

1.  **Tingkat Detail Kode:** Saya memberikan struktur *payload* JSON-RPC secara eksplisit. Ini memaksa AI untuk tidak sekadar membuat API REST biasa, tetapi benar-benar mengimplementasikan standar MCP.
2.  **Pencegahan *Cross-Environment* di Auth:** Di Fase 2, AI dipaksa memeriksa *prefix* token (`jp_sbx_` vs `jp_live_`) dan mencocokkannya dengan *header*. Ini menutup celah kebocoran data paling kritis.
3.  **Integrasi ke Decision Engine:** Di Fase 3, AI diperintahkan meneruskan `req.environment` ke `DecisionEngine.propose()`. Ini memastikan jika Agen Live melakukan request, maka *budget* dan *mandate* yang dicek adalah milik Live, bukan Sandbox.
4.  **WebMCP Context:** Di Fase 4, AI dipaksa membaca `localStorage` saat tool dieksekusi. Ini memastikan Agen Browser (seperti Gemini) selalu menembak ke environment yang sedang aktif dilihat oleh user di layar.

Dengan prompt ini, PRD 10.0 Anda akan dibangun dengan fondasi keamanan yang sama kuatnya dengan PRD 3.0 (Multi-Environment) yang sudah Anda selesaikan sebelumnya.

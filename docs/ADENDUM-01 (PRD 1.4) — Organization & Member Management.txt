## 0. ATURAN EMAS (NON-NEGOTIABLE)

1. **Backend-Configurable & Real-Time.** Semua data profil organisasi dan daftar anggota diambil dari database. Tidak ada *hardcode*.
2. **RBAC Enforcement (Strict).** Hanya `OWNER` dan `ADMIN` yang dapat mengakses halaman ini. `VIEWER`, `OPERATOR`, `APPROVER` tidak boleh melihat menu ini di sidebar.
3. **Step-Up Authentication.** Perubahan sensitif (mengubah email billing, menghapus anggota, mengubah role menjadi ADMIN) **wajib** meminta password/TOTP ulang (SCA).
4. **Immutable Audit Log.** Setiap perubahan profil org atau perubahan role anggota **wajib** menulis entri ke `compliance_audit_events` (PRD 3.0).
5. **Coverage 100%** untuk `organization_service.js` dan `member_service.js`.
6. **Zero Ambiguity.** Setiap *field*, *endpoint*, dan *test* eksplisit.

## 3. SERVICE LAYER

### 3.1 Organization Service

```javascript
// server/src/services/organization_service.js
// @ts-check
import { pool } from '../db/pool.js';
import { AppError } from '../lib/errors.js';
import { encrypt } from '../lib/crypto.js';
import { appendLedgerEntry } from './ledger.js';
import { validateStepUp } from './sca_service.js';

export async function getOrgProfile(orgId) {
  const result = await pool.query(
    `SELECT org_id, display_name, legal_name, logo_url, website_url, 
            support_email, billing_email, address, created_at, updated_at
     FROM organizations WHERE org_id = $1`,
    [orgId]
  );
  if (result.rows.length === 0) throw new AppError('ORG_NOT_FOUND', 404);
  return result.rows[0];
}

export async function updateOrgProfile(orgId, userId, body) {
  // Cek step-up jika mengubah billing_email
  if (body.billing_email) {
    await validateStepUp(userId, 'UPDATE_BILLING_EMAIL', body.step_up_grant_id);
  }

  const taxIdEnc = body.tax_id ? encrypt(body.tax_id, `organizations:tax_id_enc:${orgId}`) : null;

  const client = await pool.connect();
  try {
    await client.query('BEGIN');
    
    const result = await client.query(
      `UPDATE organizations 
       SET display_name = COALESCE($1, display_name),
           legal_name = COALESCE($2, legal_name),
           website_url = COALESCE($3, website_url),
           support_email = COALESCE($4, support_email),
           billing_email = COALESCE($5, billing_email),
           tax_id_enc = COALESCE($6, tax_id_enc),
           address = COALESCE($7, address),
           updated_at = now()
       WHERE org_id = $8 RETURNING *`,
      [body.display_name, body.legal_name, body.website_url, body.support_email, 
       body.billing_email, taxIdEnc, JSON.stringify(body.address), orgId]
    );

    // Audit Log
    await appendLedgerEntry(client, {
      orgId, type: 'ORG_PROFILE_UPDATED', subjectType: 'ORG', subjectId: orgId,
      payload: { actor: userId, fields_changed: Object.keys(body) }
    });

    await client.query('COMMIT');
    return result.rows[0];
  } catch (err) {
    await client.query('ROLLBACK');
    throw err;
  } finally {
    client.release();
  }
}
```

### 3.2 Member Service

```javascript
// server/src/services/member_service.js
// @ts-check
import { pool } from '../db/pool.js';
import crypto from 'node:crypto';
import { AppError } from '../lib/errors.js';
import { appendLedgerEntry } from './ledger.js';
import { sendInviteEmail } from './email_service.js';

export async function listMembers(orgId) {
  const result = await pool.query(
    `SELECT u.user_id, u.email, u.display_name, m.role, m.status, 
            m.last_login_at, m.joined_at
     FROM memberships m JOIN users u ON u.user_id = m.user_id
     WHERE m.org_id = $1 ORDER BY m.joined_at ASC`,
    [orgId]
  );
  return result.rows;
}

export async function inviteMember(orgId, inviterId, email, role) {
  if (role === 'OWNER') throw new AppError('CANNOT_INVITE_OWNER', 403);
  
  const token = crypto.randomBytes(32).toString('hex');
  const tokenHash = crypto.createHash('sha256').update(token).digest('hex');
  const expiresAt = new Date(Date.now() + 7 * 24 * 60 * 60 * 1000); // 7 hari

  const client = await pool.connect();
  try {
    await client.query('BEGIN');
    
    // Cek apakah email sudah menjadi member
    const existing = await client.query(
      `SELECT 1 FROM memberships m JOIN users u ON u.user_id = m.user_id 
       WHERE m.org_id = $1 AND u.email = $2`, [orgId, email]
    );
    if (existing.rows.length > 0) throw new AppError('ALREADY_MEMBER', 409);

    // Insert invitation
    const inv = await client.query(
      `INSERT INTO invitations (org_id, email, role, token_hash, invited_by, expires_at)
       VALUES ($1, $2, $3, $4, $5, $6) RETURNING invitation_id`,
      [orgId, email, role, tokenHash, inviterId, expiresAt]
    );

    // Audit Log
    await appendLedgerEntry(client, {
      orgId, type: 'MEMBER_INVITED', subjectType: 'INVITATION', subjectId: inv.rows[0].invitation_id,
      payload: { actor: inviterId, email, role }
    });

    await client.query('COMMIT');
    
    // Kirim email (async)
    const inviteUrl = `${process.env.PUBLIC_BASE_URL}/accept-invite?token=${token}`;
    await sendInviteEmail(email, inviteUrl);
    
    return { invitation_id: inv.rows[0].invitation_id, status: 'PENDING' };
  } catch (err) {
    await client.query('ROLLBACK');
    throw err;
  } finally {
    client.release();
  }
}

export async function updateMemberRole(orgId, targetUserId, newRole, actorId) {
  if (newRole === 'OWNER') throw new AppError('CANNOT_PROMOTE_TO_OWNER', 403);
  
  const client = await pool.connect();
  try {
    await client.query('BEGIN');
    
    const current = await client.query(
      `SELECT role FROM memberships WHERE org_id = $1 AND user_id = $2 FOR UPDATE`,
      [orgId, targetUserId]
    );
    if (current.rows.length === 0) throw new AppError('MEMBER_NOT_FOUND', 404);
    if (current.rows[0].role === 'OWNER') throw new AppError('CANNOT_CHANGE_OWNER_ROLE', 403);

    await client.query(
      `UPDATE memberships SET role = $1 WHERE org_id = $2 AND user_id = $3`,
      [newRole, orgId, targetUserId]
    );

    await appendLedgerEntry(client, {
      orgId, type: 'MEMBER_ROLE_UPDATED', subjectType: 'USER', subjectId: targetUserId,
      payload: { actor: actorId, old_role: current.rows[0].role, new_role: newRole }
    });

    await client.query('COMMIT');
    return { user_id: targetUserId, role: newRole };
  } catch (err) {
    await client.query('ROLLBACK');
    throw err;
  } finally {
    client.release();
  }
}
```
